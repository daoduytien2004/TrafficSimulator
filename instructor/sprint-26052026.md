### `Scene6Manager.cs`

```csharp
using System.Collections;
using UnityEngine;
using UnityEngine.SceneManagement;

public class Scene6Manager : MonoBehaviour
{
    public enum State { WaitingAtRed, AmbulanceApproaching, Completed, ViolatedRanRed, ViolatedBlocked }

    [Header("Tham chiếu")]
    public Transform playerCar;
    public Transform ambulance;
    public Transform stopLine;
    public GameObject completionPanel;   // "Hoàn thành — cách nhường đường đúng theo Điều 22"
    public GameObject ranRedPanel;       // "Vi phạm Điều 11 — vượt đèn đỏ"
    public GameObject blockedPanel;      // "Vi phạm Điều 22 — cản trở xe ưu tiên"

    [Header("Tham số")]
    public float redLightDuration = 15f;
    public float runRedThreshold = 5f;
    public float yieldSideThreshold = 0.5f;

    State currentState = State.WaitingAtRed;
    Vector3 playerStartPos;

    void Update()
    {
        if (currentState == State.AmbulanceApproaching)
            CheckBehavior();
    }

    // Gọi từ trigger zone khi xe cấp cứu đủ gần
    public void OnAmbulanceApproach()
    {
        if (currentState != State.WaitingAtRed) return;
        currentState = State.AmbulanceApproaching;
        playerStartPos = playerCar.position;
        StartCoroutine(BlockedTimer());
    }

    void CheckBehavior()
    {
        float pastLine = stopLine.position.z - playerCar.position.z;
        float sideDrift = Mathf.Abs(playerCar.position.x - playerStartPos.x);
        bool ambulancePassed = ambulance.position.z > playerCar.position.z + 3f;

        if (pastLine > runRedThreshold)
        {
            EndScenario(State.ViolatedRanRed);
            return;
        }
        if (sideDrift > yieldSideThreshold && ambulancePassed)
            EndScenario(State.Completed);
    }

    IEnumerator BlockedTimer()
    {
        yield return new WaitForSeconds(redLightDuration);
        if (currentState == State.AmbulanceApproaching)
            EndScenario(State.ViolatedBlocked);
    }

    void EndScenario(State result)
    {
        currentState = result;
        StopAllCoroutines();

        switch (result)
        {
            case State.Completed:
                completionPanel.SetActive(true);
                StartCoroutine(LoadScene("DebriefingScene", 3f));
                break;
            case State.ViolatedRanRed:
                LearningProgress.Instance?.RecordViolation(6, "Điều 11", "Vượt đèn đỏ khi nhường đường xe ưu tiên");
                ranRedPanel.SetActive(true);
                StartCoroutine(LoadScene(SceneManager.GetActiveScene().name, 4f));
                break;
            case State.ViolatedBlocked:
                LearningProgress.Instance?.RecordViolation(6, "Điều 22", "Không nhường đường cho xe ưu tiên");
                blockedPanel.SetActive(true);
                StartCoroutine(LoadScene(SceneManager.GetActiveScene().name, 4f));
                break;
        }
    }

    IEnumerator LoadScene(string sceneName, float delay)
    {
        yield return new WaitForSecondsRealtime(delay);
        SceneManager.LoadScene(sceneName);
    }
}
```

### `AmbulanceAI.cs`

```csharp
using System.Collections;
using UnityEngine;

public class AmbulanceAI : MonoBehaviour
{
    [Header("Đường đi")]
    public Transform[] waypoints;
    public float speed = 8f;

    [Header("Âm thanh & đèn")]
    public AudioSource sirenAudio;   // Spatial Blend = 1.0 trong Inspector
    public Renderer lightLeft;
    public Renderer lightRight;
    public float blinkRate = 0.3f;

    [Header("Kích hoạt kịch bản")]
    public Scene6Manager scene6Manager;
    public Transform playerCar;
    public float triggerDistance = 20f;

    int waypointIndex = 0;
    bool triggered = false;

    void Start()
    {
        sirenAudio.loop = true;
        sirenAudio.Play();
        StartCoroutine(BlinkLights());
    }

    void Update()
    {
        MoveAlongPath();
        if (!triggered && Vector3.Distance(transform.position, playerCar.position) < triggerDistance)
        {
            triggered = true;
            scene6Manager.OnAmbulanceApproach();
        }
    }

    void MoveAlongPath()
    {
        if (waypointIndex >= waypoints.Length) return;
        Transform target = waypoints[waypointIndex];
        transform.position = Vector3.MoveTowards(transform.position, target.position, speed * Time.deltaTime);
        transform.LookAt(target);
        if (Vector3.Distance(transform.position, target.position) < 0.5f)
            waypointIndex++;
    }

    IEnumerator BlinkLights()
    {
        bool toggle = false;
        while (true)
        {
            lightLeft.material.SetColor("_EmissionColor",  (toggle ? Color.blue : Color.red) * 2f);
            lightRight.material.SetColor("_EmissionColor", (toggle ? Color.red : Color.blue) * 2f);
            toggle = !toggle;
            yield return new WaitForSeconds(blinkRate);
        }
    }
}
```

### `LearningProgress.cs`

```csharp
using System.Collections.Generic;
using UnityEngine;

public class LearningProgress : MonoBehaviour
{
    public static LearningProgress Instance { get; private set; }

    public struct Violation
    {
        public int scenarioIndex;
        public string article;       // ví dụ: "Điều 15"
        public string description;   // ví dụ: "Không bật xi-nhan trước khi rẽ"
    }

    readonly List<Violation> violations = new List<Violation>();

    void Awake()
    {
        if (Instance != null) { Destroy(gameObject); return; }
        Instance = this;
        DontDestroyOnLoad(gameObject);
    }

    public void RecordViolation(int scenario, string article, string description)
    {
        violations.Add(new Violation
        {
            scenarioIndex = scenario,
            article = article,
            description = description
        });
    }

    // DebriefingScreen.cs gọi cái này để hiển thị
    public string GetReport()
    {
        if (violations.Count == 0)
            return "Không có vi phạm nào. Bạn đã hoàn thành tốt toàn bộ kịch bản!";

        var sb = new System.Text.StringBuilder();
        for (int s = 3; s <= 6; s++)
        {
            var scenarioViolations = violations.FindAll(v => v.scenarioIndex == s);
            if (scenarioViolations.Count == 0) continue;

            sb.AppendLine($"Kịch bản {s}:");
            foreach (var v in scenarioViolations)
                sb.AppendLine($"  • {v.description} → {v.article}");
        }

        sb.AppendLine();
        sb.AppendLine(GetRecommendation());
        return sb.ToString();
    }

    string GetRecommendation()
    {
        var articles = new HashSet<string>();
        foreach (var v in violations) articles.Add(v.article);

        if (articles.Count == 0) return "";
        return "Khuyến nghị: Ôn lại " + string.Join(", ", articles) + " trước khi thi sát hạch.";
    }
}
```
